3/// Tool definition for client-side function calling and server-side tools.
4///
5/// Custom tools set `name`, `description`, and `input_schema`. Server tools
6/// (web search, code execution, bash, text editor, memory, ...) set `tool_type`
7/// to a versioned identifier and a fixed `name`; use the dedicated constructors.
8#[derive(Debug, Clone, PartialEq, Deserialize)]
9pub struct Tool {
10    /// Tool type. Omitted or `"custom"` for client tools; a versioned identifier for server
11    /// tools (e.g. `web_search_20260209`, `code_execution_20260120`).
12    #[serde(rename = "type", default, skip_serializing_if = "Option::is_none")]
13    pub tool_type: Option<String>,
14    /// Tool name.
15    pub name: String,
16    /// Tool description (custom tools).
17    #[serde(default, skip_serializing_if = "Option::is_none")]
18    pub description: Option<String>,
19    /// Input schema as JSON Schema (custom tools).
20    #[serde(default, skip_serializing_if = "Option::is_none")]
21    pub input_schema: Option<serde_json::Value>,
22    /// Require schema-valid tool arguments (strict tool use).
23    #[serde(default, skip_serializing_if = "Option::is_none")]
24    pub strict: Option<bool>,
25    /// Cache control breakpoint for prompt caching.
26    #[serde(default, skip_serializing_if = "Option::is_none")]
27    pub cache_control: Option<CacheControl>,
28    /// Extra server-tool configuration fields (e.g. `max_uses`, `allowed_domains`).
29    #[serde(flatten, default)]
30    pub extra: HashMap<String, serde_json::Value>,
31}
32
33impl Tool {
34    /// Whether this is a client tool. An omitted type and the explicit `custom`
35    /// type both represent a client tool; server type tags require server execution.
36    pub fn is_client(&self) -> bool {
37        self.tool_type
38            .as_deref()
39            .is_none_or(|kind| kind == "custom")
40    }
41
42    /// Create a new custom tool definition.
43    pub fn new(
44        name: impl Into<String>,
45        description: impl Into<String>,
46        input_schema: serde_json::Value,
47    ) -> Self {
48        Self {
49            tool_type: None,
50            name: name.into(),
51            description: Some(description.into()),
52            input_schema: Some(input_schema),
53            strict: None,
54            cache_control: None,
55            extra: HashMap::new(),
56        }
57    }
58
59    /// Create a server-side tool by type + name.
60    pub fn server(tool_type: impl Into<String>, name: impl Into<String>) -> Self {
61        Self {
62            tool_type: Some(tool_type.into()),
63            name: name.into(),
64            description: None,
65            input_schema: None,
66            strict: None,
67            cache_control: None,
68            extra: HashMap::new(),
69        }
70    }
71
72    /// Built-in web search tool (`web_search_20260209`).
73    pub fn web_search() -> Self {
74        Self::server("web_search_20260209", "web_search")
75    }
76
77    /// Built-in web fetch tool (`web_fetch_20260209`).
78    pub fn web_fetch() -> Self {
79        Self::server("web_fetch_20260209", "web_fetch")
80    }
81
82    /// Built-in code execution tool (`code_execution_20260120`).
83    pub fn code_execution() -> Self {
84        Self::server("code_execution_20260120", "code_execution")
85    }
86
87    /// Built-in bash tool (`bash_20250124`).
88    pub fn bash() -> Self {
89        Self::server("bash_20250124", "bash")
90    }
91
92    /// Built-in text editor tool (`text_editor_20250728`).
93    pub fn text_editor() -> Self {
94        Self::server("text_editor_20250728", "str_replace_based_edit_tool")
95    }
96
97    /// Built-in memory tool (`memory_20250818`).
98    pub fn memory() -> Self {
99        Self::server("memory_20250818", "memory")
100    }
101
102    /// Enable strict tool use (schema-valid arguments).
103    pub fn with_strict(mut self, strict: bool) -> Self {
104        self.strict = Some(strict);
105        self
106    }
107
108    /// Attach a cache-control breakpoint to this tool.
109    pub fn with_cache_control(mut self, cache_control: CacheControl) -> Self {
110        self.cache_control = Some(cache_control);
111        self
112    }
113
114    /// Set an extra server-tool configuration field.
115    pub fn with_config(mut self, key: impl Into<String>, value: serde_json::Value) -> Self {
116        self.extra.insert(key.into(), value);
117        self
118    }
119}

121/// Tool-selection policy encoded as a tagged API object.
122#[derive(Debug, Clone, PartialEq, Default)]
123#[non_exhaustive]
124pub enum ToolChoice {
125    /// Let the model decide whether to invoke a tool.
126    #[default]
127    Auto,
128    /// Require some tool.
129    Any,
130    /// Require the named tool.
131    Tool { name: String },
132    /// Disable tool use.
133    None,
134    /// Automatic selection with an explicitly supplied parallel flag.
135    AutoWithOptions { disable_parallel_tool_use: bool },
136    /// Required selection with an explicitly supplied parallel flag.
137    AnyWithOptions { disable_parallel_tool_use: bool },
138    /// Named selection with an explicitly supplied parallel flag.
139    ToolWithOptions {
140        name: String,
141        disable_parallel_tool_use: bool,
142    },
143}
144
145impl ToolChoice {
146    /// Disable tool use.
147    pub fn none() -> Self {
148        Self::None
149    }
150    /// The canonical API discriminator, independent of optional controls.
151    pub fn kind(&self) -> &'static str {
152        match self {
153            Self::Auto | Self::AutoWithOptions { .. } => "auto",
154            Self::Any | Self::AnyWithOptions { .. } => "any",
155            Self::Tool { .. } | Self::ToolWithOptions { .. } => "tool",
156            Self::None => "none",
157        }
158    }
159    /// The forced tool's name, if any.
160    pub fn forced_tool_name(&self) -> Option<&str> {
161        match self {
162            Self::Tool { name } | Self::ToolWithOptions { name, .. } => Some(name),
163            _ => None,
164        }
165    }
166    /// Whether parallel tool use was explicitly disabled or enabled.
167    pub fn disable_parallel_tool_use(&self) -> Option<bool> {
168        match self {
169            Self::AutoWithOptions {
170                disable_parallel_tool_use,
171            }
172            | Self::AnyWithOptions {
173                disable_parallel_tool_use,
174            }
175            | Self::ToolWithOptions {
176                disable_parallel_tool_use,
177                ..
178            } => Some(*disable_parallel_tool_use),
179            _ => None,
180        }
181    }
182    /// Set the parallel control; `none` does not accept this field.
183    pub fn with_disable_parallel_tool_use(self, disabled: bool) -> crate::Result<Self> {
184        Ok(match self {
185            Self::Auto | Self::AutoWithOptions { .. } => Self::AutoWithOptions {
186                disable_parallel_tool_use: disabled,
187            },
188            Self::Any | Self::AnyWithOptions { .. } => Self::AnyWithOptions {
189                disable_parallel_tool_use: disabled,
190            },
191            Self::Tool { name } | Self::ToolWithOptions { name, .. } => Self::ToolWithOptions {
192                name,
193                disable_parallel_tool_use: disabled,
194            },
195            Self::None => {
196                return Err(crate::AnthropicError::invalid_input(
197                    "none tool choice cannot specify parallel tool use",
198                ));
199            }
200        })
201    }
202}

204impl Serialize for ToolChoice {
205    fn serialize<S: serde::Serializer>(&self, serializer: S) -> Result<S::Ok, S::Error> {
206        let mut object = serde_json::Map::new();
207        object.insert("type".into(), self.kind().into());
208        if let Some(name) = self.forced_tool_name() {
209            if name.is_empty() {
210                return Err(serde::ser::Error::custom(
211                    "forced tool name cannot be empty",
212                ));
213            }
214            object.insert("name".into(), name.into());
215        }
216        if let Some(disabled) = self.disable_parallel_tool_use() {
217            object.insert("disable_parallel_tool_use".into(), disabled.into());
218        }
219        object.serialize(serializer)
220    }
221}
222impl<'de> Deserialize<'de> for ToolChoice {
223    fn deserialize<D: serde::Deserializer<'de>>(deserializer: D) -> Result<Self, D::Error> {
224        let value = Value::deserialize(deserializer)?;
225        // Historic null cannot distinguish Auto from Any; Auto is deterministic.
226        if value.is_null() {
227            return Ok(Self::Auto);
228        }
229        let object = value
230            .as_object()
231            .ok_or_else(|| serde::de::Error::custom("tool choice must be an object"))?;
232        let choice = parse_tool_choice(object).map_err(serde::de::Error::custom)?;
233        match object.get("disable_parallel_tool_use") {
234            None => Ok(choice),
235            Some(Value::Bool(disabled)) => choice
236                .with_disable_parallel_tool_use(*disabled)
237                .map_err(serde::de::Error::custom),
238            _ => Err(serde::de::Error::custom(
239                "parallel tool-use flag must be a boolean",
240            )),
241        }
242    }
243}
244
245fn parse_tool_choice(object: &serde_json::Map<String, Value>) -> Result<ToolChoice, &'static str> {
246    if object
247        .keys()
248        .any(|key| !["type", "name", "disable_parallel_tool_use"].contains(&key.as_str()))
249    {
250        return Err("unexpected tool-choice field");
251    }
252    let kind = match object.get("type") {
253        Some(Value::String(kind)) => kind.as_str(),
254        None if object.len() == 1 && object.contains_key("name") => "tool",
255        _ => return Err("tool choice requires a string type"),
256    };
257    let name = object.get("name");
258    match kind {
259        "auto" if name.is_none() => Ok(ToolChoice::Auto),
260        "any" if name.is_none() => Ok(ToolChoice::Any),
261        "none" if name.is_none() => Ok(ToolChoice::None),
262        "tool" => Ok(ToolChoice::Tool {
263            name: name
264                .and_then(Value::as_str)
265                .filter(|name| !name.is_empty())
266                .ok_or("tool choice requires a nonempty name")?
267                .to_owned(),
268        }),
269        _ => Err("invalid tool choice type or fields"),
270    }
271}

273impl Serialize for Tool {
274    fn serialize<S: serde::Serializer>(&self, serializer: S) -> Result<S::Ok, S::Error> {
275        let mut fields = vec![("name", Value::String(self.name.clone()))];
276        if let Some(value) = &self.tool_type {
277            fields.push(("type", Value::String(value.clone())));
278        }
279        if let Some(value) = &self.description {
280            fields.push(("description", Value::String(value.clone())));
281        }
282        if let Some(value) = &self.input_schema {
283            fields.push(("input_schema", value.clone()));
284        }
285        if let Some(value) = self.strict {
286            fields.push(("strict", Value::Bool(value)));
287        }
288        if let Some(value) = &self.cache_control {
289            fields.push((
290                "cache_control",
291                serde_json::to_value(value).map_err(serde::ser::Error::custom)?,
292            ));
293        }
294        serialize_object(
295            serializer,
296            None,
297            fields,
298            &self.extra,
299            &[
300                "type",
301                "name",
302                "description",
303                "input_schema",
304                "strict",
305                "cache_control",
306            ],
307        )
308    }
