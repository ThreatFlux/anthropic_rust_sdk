3/// Stop reason enumeration
4#[derive(Debug, Clone, PartialEq, Eq)]
5#[non_exhaustive]
6pub enum StopReason {
7    /// Hit maximum tokens limit
8    MaxTokens,
9    /// Natural end of message
10    EndTurn,
11    /// Stop sequence encountered
12    StopSequence,
13    /// Tool use requested
14    ToolUse,
15    /// Model paused and expects the conversation to continue
16    PauseTurn,
17    /// Response was declined for safety/policy reasons
18    Refusal,
19    /// Model context window was exhausted.
20    ModelContextWindowExceeded,
21    /// An unfamiliar API reason, preserved exactly.
22    Unknown(String),
23}
24
25/// Structured detail accompanying a `refusal` (and other) stop reason.
26#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, Default)]
27pub struct StopDetails {
28    /// Detail type (e.g. `"refusal"`).
29    #[serde(rename = "type", default, skip_serializing_if = "Option::is_none")]
30    pub detail_type: Option<String>,
31    /// Policy category, e.g. `"cyber"`, `"bio"`, `"reasoning_extraction"`.
32    #[serde(default, skip_serializing_if = "Option::is_none")]
33    pub category: Option<String>,
34    /// Human-readable explanation, when provided.
35    #[serde(default, skip_serializing_if = "Option::is_none")]
36    pub explanation: Option<String>,
37    /// Recommended model to retry with, when provided.
38    #[serde(default, skip_serializing_if = "Option::is_none")]
39    pub recommended_model: Option<String>,
40    /// Opaque token used to redeem fallback credit on a retry.
41    #[serde(default, skip_serializing_if = "Option::is_none")]
42    pub fallback_credit_token: Option<String>,
43    /// Whether the refusal token can be redeemed with assistant-prefill form.
44    #[serde(default, skip_serializing_if = "Option::is_none")]
45    pub fallback_has_prefill_claim: Option<bool>,
46    /// Forward-compatible extra fields.
47    #[serde(flatten, default)]
48    pub extra: HashMap<String, serde_json::Value>,
49}
50impl StopReason {
51    /// The exact API wire value.
52    pub fn as_str(&self) -> &str {
53        match self {
54            Self::MaxTokens => "max_tokens",
55            Self::EndTurn => "end_turn",
56            Self::StopSequence => "stop_sequence",
57            Self::ToolUse => "tool_use",
58            Self::PauseTurn => "pause_turn",
59            Self::Refusal => "refusal",
60            Self::ModelContextWindowExceeded => "model_context_window_exceeded",
61            Self::Unknown(value) => value,
62        }
63    }
64}
