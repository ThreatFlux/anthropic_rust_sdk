3/// Cache control for prompt caching.
4///
5/// Attach to a content block, tool, or system block to mark a cache breakpoint,
6/// or set [`crate::models::message::MessageRequest::cache_control`] to auto-cache
7/// the last cacheable block.
8#[derive(Debug, Clone, PartialEq, Eq, Deserialize)]
9pub struct CacheControl {
10    /// Type of cache control (always `"ephemeral"`).
11    #[serde(rename = "type")]
12    pub cache_type: String,
13    /// Time-to-live for the cache entry: `"5m"` (default) or `"1h"`.
14    #[serde(default, skip_serializing_if = "Option::is_none")]
15    pub ttl: Option<String>,
16    /// Forward-compatible protocol fields. Reserved typed keys cannot be overridden.
17    #[serde(flatten, default)]
18    pub extra: ExtraFields,
19}
20
21impl CacheControl {
22    /// Create ephemeral cache control with the default 5-minute TTL.
23    pub fn ephemeral() -> Self {
24        Self {
25            cache_type: "ephemeral".to_string(),
26            ttl: None,
27            extra: ExtraFields::new(),
28        }
29    }
30
31    /// Create ephemeral cache control with a 1-hour TTL.
32    pub fn ephemeral_1h() -> Self {
33        Self {
34            cache_type: "ephemeral".to_string(),
35            ttl: Some("1h".to_string()),
36            extra: ExtraFields::new(),
37        }
38    }
39}
40impl Serialize for CacheControl {
41    fn serialize<S: serde::Serializer>(&self, serializer: S) -> Result<S::Ok, S::Error> {
42        #[allow(unused_mut)]
43        let mut fields = vec![(
44            "type",
45            serde_json::to_value(&self.cache_type).map_err(serde::ser::Error::custom)?,
46        )];
47        if let Some(value) = &self.ttl {
48            fields.push((
49                "ttl",
50                serde_json::to_value(value).map_err(serde::ser::Error::custom)?,
51            ));
52        }
53        serialize_object(serializer, None, fields, &self.extra, &["type", "ttl"])
54    }
55}
