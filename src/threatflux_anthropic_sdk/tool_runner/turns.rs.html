3struct TurnState {
4    seen_ids: HashSet<String>,
5    offered: HashSet<String>,
6}
7impl TurnState {
8    fn new(request: &MessageRequest) -> Self {
9        let seen_ids = request
10            .messages
11            .iter()
12            .flat_map(|message| &message.content)
13            .filter_map(|block| match block {
14                ContentBlock::ToolUse { id, .. } => Some(id.clone()),
15                _ => None,
16            })
17            .collect();
18        let offered = request
19            .tools
20            .as_deref()
21            .unwrap_or_default()
22            .iter()
23            .filter(|tool| tool.is_client())
24            .map(|tool| tool.name.clone())
25            .collect();
26        Self { seen_ids, offered }
27    }
28}

30impl ToolRunner {
31    pub(super) async fn run_loop(
32        &self,
33        mut request: MessageRequest,
34        options: Option<RequestOptions>,
35        streaming: bool,
36        result: &mut ToolRunnerResult,
37        started: &StartedCalls,
38    ) -> ToolRunnerTermination {
39        let mut state = TurnState::new(&request);
40        for _ in 0..self.options.max_turns {
41            if self.cancellation.is_cancelled() {
42                return ToolRunnerTermination::Cancelled;
43            }
44            request.messages = result.messages.clone();
45            if let Err(failure) = self
46                .run_turn(
47                    &request,
48                    options.clone(),
49                    streaming,
50                    result,
51                    started,
52                    &mut state,
53                )
54                .await
55            {
56                return result.stop(failure);
57            }
58        }
59        ToolRunnerTermination::TurnLimit
60    }
61
62    async fn run_turn(
63        &self,
64        request: &MessageRequest,
65        options: Option<RequestOptions>,
66        streaming: bool,
67        result: &mut ToolRunnerResult,
68        started: &StartedCalls,
69        state: &mut TurnState,
70    ) -> RunResult<()> {
71        let response = self
72            .receive_turn(request.clone(), options, streaming)
73            .await
74            .map_err(|error| (ToolRunnerTermination::TransportFailed, Some(error)))?;
75        result.record_response(&response);
76        if let Some(termination) = response_termination(response.stop_reason.as_ref()) {
77            return Err((termination, None));
78        }
79        let prepared = self.prepare_turn(request, &response, state, started)?;
80        self.approve_calls(&prepared).await?;
81        self.execute_calls(prepared, request.tool_choice.as_ref(), result, started)
82            .await
83    }
84
85    async fn receive_turn(
86        &self,
87        request: MessageRequest,
88        options: Option<RequestOptions>,
89        streaming: bool,
90    ) -> Result<MessageResponse> {
91        if streaming {
92            self.client
93                .messages()
94                .create_stream(request, options)
95                .await?
96                .collect_message()
97                .await
98        } else {
99            self.client.messages().create(request, options).await
100        }
101    }
102
103    fn prepare_turn(
104        &self,
105        request: &MessageRequest,
106        response: &MessageResponse,
107        state: &mut TurnState,
108        started: &StartedCalls,
109    ) -> RunResult<PreparedCalls> {
110        if response
111            .content
112            .iter()
113            .any(|block| matches!(block, ContentBlock::Compaction { .. }))
114            && self.options.compaction == CompactionReplayPolicy::Reject
115        {
116            return Err((ToolRunnerTermination::UnsupportedReplay, None));
117        }
118        response
119            .to_conversation_message(self.options.unknown_content)
120            .map_err(|error| (ToolRunnerTermination::UnsupportedReplay, Some(error)))?;
121        let calls = client_calls(response);
122        if calls.is_empty() {
123            return Err((ToolRunnerTermination::NoClientToolCalls, None));
124        }
125        let started_count = started.lock().expect("tool-start ledger poisoned").len();
126        if calls.len() > self.options.max_tool_calls.saturating_sub(started_count) {
127            return Err((ToolRunnerTermination::CallLimit, None));
128        }
129        self.prepare_calls(calls, request.tool_choice.as_ref(), state)
130    }
131
132    fn prepare_calls(
133        &self,
134        calls: Vec<ToolCall>,
135        choice: Option<&ToolChoice>,
136        state: &mut TurnState,
137    ) -> RunResult<PreparedCalls> {
138        let mut prepared = Vec::with_capacity(calls.len());
139        for call in calls {
140            validate_call_allowed(&call, choice, state)
141                .map_err(|error| (ToolRunnerTermination::InvalidToolCall, Some(error)))?;
142            let tool = self
143                .registry
144                .tools
145                .get(&call.name)
146                .expect("offered registered tool")
147                .clone();
148            (tool.validator)(&call.input)
149                .map_err(|error| (ToolRunnerTermination::InvalidToolCall, Some(error)))?;
150            prepared.push((call, tool));
151        }
152        Ok(prepared)
153    }
154}
155
156fn client_calls(response: &MessageResponse) -> Vec<ToolCall> {
157    response
158        .content
159        .iter()
160        .filter_map(|block| match block {
161            ContentBlock::ToolUse {
162                id, name, input, ..
163            } => Some(ToolCall {
164                id: id.clone(),
165                name: name.clone(),
166                input: input.clone(),
167            }),
168            _ => None,
169        })
170        .collect()
171}
172
173fn validate_call_allowed(
174    call: &ToolCall,
175    choice: Option<&ToolChoice>,
176    state: &mut TurnState,
177) -> Result<()> {
178    if call.id.is_empty()
179        || !state.seen_ids.insert(call.id.clone())
180        || !state.offered.contains(&call.name)
181        || choice.is_some_and(|choice| choice.kind() == "none")
182        || choice
183            .and_then(ToolChoice::forced_tool_name)
184            .is_some_and(|name| name != call.name)
185    {
186        return Err(AnthropicError::invalid_input(
187            "unknown, disabled, duplicate, or mismatched tool invocation",
188        ));
189    }
190    Ok(())
191}
192
193fn response_termination(reason: Option<&StopReason>) -> Option<ToolRunnerTermination> {
194    Some(match reason {
195        Some(StopReason::ToolUse) => return None,
196        Some(StopReason::EndTurn) => ToolRunnerTermination::EndTurn,
197        Some(StopReason::StopSequence) => ToolRunnerTermination::StopSequence,
198        Some(StopReason::Refusal) => ToolRunnerTermination::Refusal,
199        Some(StopReason::MaxTokens) => ToolRunnerTermination::MaxTokens,
200        Some(StopReason::PauseTurn) => ToolRunnerTermination::PauseTurn,
201        Some(StopReason::ModelContextWindowExceeded) => {
202            ToolRunnerTermination::ContextWindowExceeded
203        }
204        Some(StopReason::Unknown(reason)) => {
205            ToolRunnerTermination::UnknownStopReason(reason.clone())
206        }
207        None => ToolRunnerTermination::MissingStopReason,
208    })
