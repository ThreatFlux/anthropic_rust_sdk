3/// A requested client invocation supplied to an optional execution hook.
4#[derive(Debug, Clone)]
5#[non_exhaustive]
6pub struct ToolCall {
7    /// Original identifier used in the corresponding tool result.
8    pub id: String,
9    /// Explicitly registered name.
10    pub name: String,
11    /// Complete, already decoded object input.
12    pub input: Value,
13}

15/// Callback error handling. Remote error content never includes callback details.
16#[derive(Debug, Clone, Copy, PartialEq, Eq)]
17#[non_exhaustive]
18pub enum CallbackErrorPolicy {
19    /// Stop and return the partial transcript plus the local callback error.
20    Stop,
21    /// Return fixed sanitized `is_error` content to the model, then continue.
22    ReturnSanitizedError,
23}
24/// How a complete compaction block is replayed on a subsequent turn.
25#[derive(Debug, Clone, Copy, PartialEq, Eq)]
26#[non_exhaustive]
27pub enum CompactionReplayPolicy {
28    /// Stop before callbacks; the caller decides how to continue compacted history.
29    Reject,
30    /// Keep the full append-only history and compaction block verbatim. Never
31    /// manufacture a summary or silently discard messages before the boundary.
32    PreserveHistory,
33}

35/// Finite execution limits. Start from `Default` and change individual fields.
36#[derive(Debug, Clone)]
37#[non_exhaustive]
38pub struct ToolRunnerOptions {
39    /// Maximum Messages API turns.
40    pub max_turns: usize,
41    /// Maximum callbacks started across the whole run.
42    pub max_tool_calls: usize,
43    /// Deadline for an individual execution hook or callback.
44    pub call_timeout: Duration,
45    /// Deadline for all HTTP, hook, and callback work.
46    pub overall_timeout: Duration,
47    /// Maximum serialized tool-result content bytes per call.
48    pub max_result_bytes: usize,
49    /// Maximum concurrently executing callbacks; defaults to one.
50    pub max_parallel_calls: usize,
51    /// Callback failure handling; defaults to returning a partial result.
52    pub callback_errors: CallbackErrorPolicy,
53    /// Unknown content replay requires an explicit preservation choice.
54    pub unknown_content: ReplayUnknownPolicy,
55    /// Explicit handling for server-produced compaction snapshots.
56    pub compaction: CompactionReplayPolicy,
57}
58impl Default for ToolRunnerOptions {
59    fn default() -> Self {
60        Self {
61            max_turns: 20,
62            max_tool_calls: 100,
63            call_timeout: Duration::from_secs(30),
64            overall_timeout: Duration::from_secs(300),
65            max_result_bytes: 1024 * 1024,
66            max_parallel_calls: 1,
67            callback_errors: CallbackErrorPolicy::Stop,
68            unknown_content: ReplayUnknownPolicy::Reject,
69            compaction: CompactionReplayPolicy::Reject,
70        }
71    }
72}
73impl ToolRunnerOptions {
74    /// Reject zero or overflowing budgets before constructing a runner.
75    pub fn validate(&self) -> Result<()> {
76        if self.max_turns == 0
77            || self.max_tool_calls == 0
78            || self.call_timeout.is_zero()
79            || self.overall_timeout.is_zero()
80            || self.max_result_bytes == 0
81            || self.max_parallel_calls == 0
82        {
83            return Err(AnthropicError::invalid_input(
84                "tool-runner budgets must be nonzero",
85            ));
86        }
87        if tokio::time::Instant::now()
88            .checked_add(self.overall_timeout)
89            .is_none()
90            || tokio::time::Instant::now()
91                .checked_add(self.call_timeout)
92                .is_none()
93        {
94            return Err(AnthropicError::invalid_input(
95                "tool-runner deadline exceeds supported duration",
96            ));
97        }
98        Ok(())
99    }
