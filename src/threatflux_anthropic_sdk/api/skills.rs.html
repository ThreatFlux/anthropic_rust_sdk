37    /// Build multipart form payload for skill upload operations.
38    fn build_skill_upload_form(
39        display_title: Option<&str>,
40        files: Vec<SkillFileUpload>,
41    ) -> Result<Form> {
42        let mut form = Form::new();
43
44        if let Some(display_title) = display_title {
45            form = form.text("display_title", display_title.to_string());
46        }
47
48        for file in files {
49            let part = Part::bytes(file.content)
50                .file_name(file.filename)
51                .mime_str(&file.mime_type)
52                .map_err(|e| {
53                    AnthropicError::file_error(format!("Invalid MIME type for skill file: {}", e))
54                })?;
55            form = form.part("files", part);
56        }
57
58        Ok(form)
59    }

77    /// Convert a local directory into skill upload files.
78    fn collect_dir_files(root: &Path) -> Result<Vec<std::path::PathBuf>> {
79        let root_metadata = std::fs::symlink_metadata(root).map_err(|e| {
80            AnthropicError::file_error(format!(
81                "Failed to read directory metadata {}: {}",
82                root.display(),
83                e
84            ))
85        })?;
86
87        if root_metadata.file_type().is_symlink() {
88            return Err(AnthropicError::file_error(format!(
89                "Symlinks are not allowed in skill directories: {}",
90                root.display()
91            )));
92        }
93
94        if !root.exists() {
95            return Err(AnthropicError::file_error(format!(
96                "Directory does not exist: {}",
97                root.display()
98            )));
99        }
100        if !root.is_dir() {
101            return Err(AnthropicError::file_error(format!(
102                "Path is not a directory: {}",
103                root.display()
104            )));
105        }
106
107        let mut files = Vec::new();
108        let mut stack = vec![root.to_path_buf()];
109
110        while let Some(dir) = stack.pop() {
111            let entries = std::fs::read_dir(&dir).map_err(|e| {
112                AnthropicError::file_error(format!(
113                    "Failed to read directory {}: {}",
114                    dir.display(),
115                    e
116                ))
117            })?;
118
119            for entry in entries {
120                let entry = entry.map_err(|e| {
121                    AnthropicError::file_error(format!("Failed to read directory entry: {}", e))
122                })?;
123                let path = entry.path();
124                let file_type = entry.file_type().map_err(|e| {
125                    AnthropicError::file_error(format!(
126                        "Failed to read file type for {}: {}",
127                        path.display(),
128                        e
129                    ))
130                })?;
131
132                if file_type.is_symlink() {
133                    return Err(AnthropicError::file_error(format!(
134                        "Symlinks are not allowed in skill directories: {}",
135                        path.display()
136                    )));
137                }
138
139                if file_type.is_dir() {
140                    stack.push(path);
141                } else if file_type.is_file() {
142                    files.push(path);
143                }
144            }
145        }
146
147        files.sort();
148        Ok(files)
149    }

151    /// Build skill upload files from a local directory.
152    async fn build_upload_files_from_dir(root: &Path) -> Result<Vec<SkillFileUpload>> {
153        let all_paths = Self::collect_dir_files(root)?;
154        if all_paths.is_empty() {
155            return Err(AnthropicError::invalid_input(format!(
156                "No files found in directory: {}",
157                root.display()
158            )));
159        }
160
161        let root_name = root.file_name().ok_or_else(|| {
162            AnthropicError::invalid_input(format!(
163                "Skill directory path must have a final directory name: {}",
164                root.display()
165            ))
166        })?;
167
168        let mut files = Vec::with_capacity(all_paths.len());
169
170        for path in all_paths {
171            let rel = path.strip_prefix(root).map_err(|e| {
172                AnthropicError::file_error(format!(
173                    "Failed to compute relative path for {}: {}",
174                    path.display(),
175                    e
176                ))
177            })?;
178            let remote_path = Path::new(root_name).join(rel);
179            let remote_filename = remote_path.to_string_lossy().replace('\\', "/");
180            let content = tokio::fs::read(&path).await.map_err(|e| {
181                AnthropicError::file_error(format!("Failed to read file {}: {}", path.display(), e))
182            })?;
183            let mime_type = mime_guess::from_path(&path)
184                .first_or_octet_stream()
185                .to_string();
186
187            files.push(SkillFileUpload::new(remote_filename, content, mime_type));
188        }
189
190        Ok(files)
191    }

468    /// Lazily traverse legacy Skills pages while preserving filters and options.
469    pub fn pages(
470        &self,
471        mut params: SkillListParams,
472        limits: PaginationLimits,
473        options: Option<RequestOptions>,
474    ) -> Result<PageStream<Skill>> {
475        validate_skill_pagination(params.limit, params.page.as_deref())?;
476        params.limit = params.limit.or(Some(100));
477        let initial = params.page.take();
478        let api = self.clone();
479        paginate(limits, initial, move |cursor| {
480            let api = api.clone();
481            let mut params = params.clone();
482            let options = options.clone();
483            params.page = cursor;
484            async move {
485                let response = api.list(Some(params), options).await?;
486                let next_cursor = legacy_skill_cursor(response.has_more, response.next_page)?;
487                let item_ids = response.data.iter().map(|skill| skill.id.clone()).collect();
488                Ok(TraversalPage {
489                    data: response.data,
490                    next_cursor,
491                    item_ids,
492                })
493            }
494        })
495    }

509        &self,

557fn validate_skill_pagination(limit: Option<u32>, page: Option<&str>) -> Result<()> {
558    if limit.is_some_and(|value| !(1..=1000).contains(&value)) {
559        return Err(AnthropicError::invalid_input(
560            "Skill page size must be between 1 and 1000",
561        ));
562    }
563    if page == Some("") {
564        return Err(AnthropicError::invalid_input(
565            "Skill page token must not be empty",
566        ));
567    }
568    Ok(())
569}
570
571fn legacy_skill_cursor(has_more: bool, next_page: Option<String>) -> Result<Option<String>> {
572    if has_more && next_page.is_none() {
573        return Err(AnthropicError::invalid_input(
574            "Legacy Skills response has_more without next_page",
575        ));
576    }
577    // Token cursors themselves determine continuation, even if has_more was omitted.
578    Ok(next_page)
579}

581fn current_skill_options(options: Option<RequestOptions>) -> Result<Option<RequestOptions>> {
582    if let Some(options) = &options {
583        let conflicting_header = options.headers.iter().any(|(name, value)| {
584            name.eq_ignore_ascii_case("anthropic-beta")
585                && value
586                    .split(',')
587                    .any(|beta| beta.trim() == beta_headers::SKILLS_API)
588        });
589        if options.enable_skills_api
590            || options.beta_features.iter().any(|entry| {
591                entry
592                    .split(',')
593                    .any(|beta| beta.trim() == beta_headers::SKILLS_API)
594            })
595            || conflicting_header
596        {
597            return Err(AnthropicError::invalid_input(
598                "Current Skills cannot use skills-2025-10-02; use skills_legacy() for that schema",
599            ));
600        }
601    }
602    Ok(options)
603}

605fn skill_path_segment(id: &str) -> Result<String> {
606    if id.is_empty() || matches!(id, "." | "..") {
607        return Err(AnthropicError::invalid_input(
608            "Skill and version IDs must not be empty or dot path segments",
609        ));
610    }
611    Ok(id
612        .bytes()
613        .map(|byte| {
614            if byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'_' | b'.' | b'~') {
615                char::from(byte).to_string()
616            } else {
617                format!("%{byte:02X}")
618            }
619        })
620        .collect())
621}

848    /// Traverse current skill pages lazily, preserving source and request options.
849    pub fn pages(
850        &self,
851        mut params: SkillListParams,
852        limits: PaginationLimits,
853        options: Option<RequestOptions>,
854    ) -> Result<PageStream<CurrentSkill>> {
855        validate_skill_pagination(params.limit, params.page.as_deref())?;
856        current_skill_options(options.clone())?;
857        params.limit = params.limit.or(Some(100));
858        let initial = params.page.take();
859        let api = self.clone();
860        paginate(limits, initial, move |cursor| {
861            let api = api.clone();
862            let mut params = params.clone();
863            let options = options.clone();
864            params.page = cursor;
865            async move {
866                let response = api.list(Some(params), options).await?;
867                let item_ids = response.data.iter().map(|skill| skill.id.clone()).collect();
868                Ok(TraversalPage {
869                    data: response.data,
870                    next_cursor: response.next_page,
871                    item_ids,
872                })
873            }
874        })
875    }

900    ) -> Result<PageStream<CurrentSkillVersion>> {

902        skill_path_segment(skill_id)?;

904        params.limit = params.limit.or(Some(100));
905        let initial = params.page.take();
906        let api = self.clone();
907        let skill_id = skill_id.to_string();
908        paginate(limits, initial, move |cursor| {
909            let api = api.clone();
910            let skill_id = skill_id.clone();
911            let mut params = params.clone();
912            let options = options.clone();
913            params.page = cursor;
914            async move {

916                let item_ids = response
917                    .data
918                    .iter()
919                    .map(|version| version.id.clone())
920                    .collect();
921                Ok(TraversalPage {
922                    data: response.data,
923                    next_cursor: response.next_page,
924                    item_ids,
925                })
926            }
927        })
